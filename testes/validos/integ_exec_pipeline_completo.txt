def bonus(pontos) {
    if pontos >= 50 {
        return 15;
    } elif pontos >= 25 {
        return 5;
    }
    return 0;
}

def calcular_final(base, b) {
    return (base + b) * 2 - 3;
}

escores = [10, 30, -5, 55, 25, 90];
processados = [0, 0, 0, 0];
qtd = 0;

for s in escores {
    if s < 0 {
        continue;
    }
    if qtd == 4 {
        break;
    }
    b = bonus(s);
    processados[qtd] = calcular_final(s, b);
    qtd += 1;
}

acc = 0;
idx = 0;
while idx < qtd {
    acc += processados[idx];
    idx += 1;
}

print(acc);
